66      // Завдання 5. Один SaveChangesAsync для масиву; відповідь містить усі створені Id.
67      [HttpPost("batch")]
68      public async Task<ActionResult<IEnumerable<TodoItemDto>>> PostTodoItems(TodoItemDto[] items)
69      {
70          if (items.Length == 0 || items.Length > 200)
71              return BadRequest(new { error = "Масив має містити від 1 до 200 елементів." });
72          if (items.Any(item => item is null || item.Id != 0))
73              return BadRequest(new { error = "Кожен елемент має бути об'єктом без заданого Id." });
74          var created = items.Select(FromDto).ToArray();
75          context.TodoItems.AddRange(created);
76          await context.SaveChangesAsync();
77          return CreatedAtAction(nameof(GetTodoItems), created.Select(ToDto).ToArray());
78      }
